% --- Grafica, figure e resa tipografica ---
% NOTA: expansion=false necessario con fontenc T1 per evitare il fatal error
%   "auto expansion is only possible with scalable fonts" di pdfTeX/microtype.
%   La protrusion (margini ottici) rimane attiva; solo l'espansione del font
%   (che richiede font scalabili Type1/OTF) è disabilitata.
\usepackage[protrusion=true, expansion=false]{microtype}
\usepackage{xcolor}
\usepackage{graphicx}
\usepackage{float}
\usepackage{subfigure}
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta, positioning, shapes.geometric, calc}

\graphicspath{{../Images/}{../Rampa/}{../Paletta/}}
% emergencystretch: allarga la spaziatura inter-parola come ultima risorsa

% --- Codice sorgente ---
\usepackage{listings}

\definecolor{codegreen}{rgb}{0.0,0.55,0.0}
\definecolor{codegray}{rgb}{0.5,0.5,0.5}
\definecolor{codepurple}{rgb}{0.58,0,0.82}
\definecolor{backcolour}{rgb}{0.97,0.97,0.97}
\definecolor{keyblue}{rgb}{0.0,0.35,0.75}

% Parametri comuni a tutti gli stili di codice: garantiscono che il riquadro sia
% staccato dai margini della pagina (xleftmargin/xrightmargin) e che il testo non
% tocchi la cornice (framesep), migliorando la leggibilita' del codice ovunque.
\lstdefinestyle{base}{
    backgroundcolor=\color{backcolour},
    commentstyle=\color{codegreen}\itshape,
    keywordstyle=\color{keyblue}\bfseries,
    numberstyle=\tiny\color{codegray},
    stringstyle=\color{codepurple},
    basicstyle=\ttfamily\footnotesize,
    breakatwhitespace=false,
    breaklines=true,
    postbreak=\mbox{\textcolor{codegray}{$\hookrightarrow$}\space},
    captionpos=b,
    columns=fullflexible,
    keepspaces=true,
    numbers=left,
    numbersep=8pt,
    showspaces=false,
    showstringspaces=false,
    showtabs=false,
    tabsize=2,
    frame=single,
    framesep=6pt,
    xleftmargin=1.8em,
    xrightmargin=1.2em,
    aboveskip=1.2em,
    belowskip=1.2em,
    rulecolor=\color{gray!40},
}

\lstdefinestyle{gmsh}{
    style=base,
    morekeywords={Point, Line, Spline, Line Loop, Plane Surface, Physical Surface,
                  Physical Line, Transfinite Line, Transfinite Surface,
                  Recombine, Field, Background, Using, Progression, Bump,
                  Attractor, Threshold, EdgesList, EdgeList, IField,
                  LcMin, LcMax, DistMin, DistMax, Mesh, MshFileVersion,
                  Periodic, SetFactory},
}

\lstdefinestyle{fortran}{
    style=base,
    language=[90]Fortran,
}
